### Priority 1 — Croupier expression-token evaluation

`backend/openshiksha/apps/api/croupier.py`:

```python
import math
import re
import ast
import operator as _op

_TOKEN_RE = re.compile(r"\{\{([^{}]+)\}\}")

_SAFE_BIN = {
    ast.Add: _op.add, ast.Sub: _op.sub,
    ast.Mult: _op.mul, ast.Div: _op.truediv,
    ast.Pow: _op.pow, ast.Mod: _op.mod,
}
_SAFE_UNARY = {ast.USub: _op.neg, ast.UAdd: _op.pos}

_SAFE_CONSTS = {
    "pi_val": math.pi,
    "e_val":  math.e,
}

def _trunc(x, n=0):
    factor = 10 ** int(n)
    return math.trunc(float(x) * factor) / factor

_SAFE_FUNCS = {
    "trunc":   _trunc,
    "round":   round,
    "abs":     abs,
    "min":     min,
    "max":     max,
    "sqrt":    math.sqrt,
    "floor":   math.floor,
    "ceil":    math.ceil,
    "Decimal": float,  # legacy wrapper — modern is float-native
    "int":     int,
    "float":   float,
}

def substitute_variables(text: str, variable_values: dict) -> str:
    """
    Replace every {{...}} token in text with its evaluated value.

    A bare identifier maps directly to the sampled variable.
    Anything else (`{{j*2}}`, `{{trunc(pi_val*r*r, 2)}}`) is evaluated
    via the safe AST evaluator with the sampled values in scope.
    A token that fails to evaluate is left intact rather than raising,
    so a single bad expression never blanks an entire question.
    """
    if not text:
        return text or ""

    def _repl(m: re.Match[str]) -> str:
        expr = m.group(1).strip()
        if expr.isidentifier() and expr in variable_values:
            return _fmt(variable_values[expr])
        try:
            return _fmt(_eval_expr(expr, variable_values))
        except Exception:
            return m.group(0)

    return _TOKEN_RE.sub(_repl, text)

def _fmt(value) -> str:
    if isinstance(value, bool):  # bool is an int subclass — handle first
        return str(value)
    if isinstance(value, float):
        if value.is_integer():
            return str(int(value))
        return f"{value:.4f}".rstrip("0").rstrip(".")
    return str(value)

def _eval_expr(expr: str, variable_values: dict) -> float:
    tree = ast.parse(expr, mode="eval")
    return _eval_node(tree.body, variable_values)

def _eval_node(node, vars_):
    if isinstance(node, ast.Constant) and isinstance(node.value, (int, float)):
        return node.value
    if isinstance(node, ast.Name):
        if node.id in vars_:
            return vars_[node.id]
        if node.id in _SAFE_CONSTS:
            return _SAFE_CONSTS[node.id]
        raise ValueError(f"unknown identifier {node.id!r}")
    if isinstance(node, ast.BinOp) and type(node.op) in _SAFE_BIN:
        return _SAFE_BIN[type(node.op)](_eval_node(node.left, vars_), _eval_node(node.right, vars_))
    if isinstance(node, ast.UnaryOp) and type(node.op) in _SAFE_UNARY:
        return _SAFE_UNARY[type(node.op)](_eval_node(node.operand, vars_))
    if isinstance(node, ast.Call) and isinstance(node.func, ast.Name) and node.func.id in _SAFE_FUNCS:
        args = [_eval_node(a, vars_) for a in node.args]
        kwargs = {kw.arg: _eval_node(kw.value, vars_) for kw in node.keywords if kw.arg}
        return _SAFE_FUNCS[node.func.id](*args, **kwargs)
    raise ValueError(f"unsupported node {type(node).__name__}")

# safe_eval_expr keeps its old name for backwards compat — now delegates.
def safe_eval_expr(expr: str, variable_values: dict) -> float:
    return float(_eval_expr(substitute_variables(expr, variable_values), variable_values))
```

# was r"_\{(\{[^{}]+\})\}_" / r"\1"  — stripped a brace pair, breaking 751 subparts
_EXPR_TOKEN = re.compile(r"_\{\{([^{}]+)\}\}_")

def convert_tokens(text: str) -> str:
    if not text:
        return text or ""
    text = _EXPR_TOKEN.sub(r"{{\1}}", text)
    text = _VAR_TOKEN.sub(r"{{\1}}", text)
    return text

### Priority 3 — Serializer: substitute solutions & hints per student

`backend/openshiksha/apps/api/serializers/core.py` —
`QuestionWithSubpartsStudentSerializer.to_representation`:

```python
from openshiksha.apps.api.croupier import (
    substitute_variables,
    substitute_variables_for_student,
)

def to_representation(self, instance):
    data = super().to_representation(instance)
    student = self.context["request"].user
    for sp in data["subparts"]:
        text, options, values = substitute_variables_for_student(
            sp["question_text"], sp["options"],
            sp["variable_constraints"], student.id, sp["id"],
        )
        sp["question_text"] = text
        sp["options"] = options
        # NEW — same seeded values, so solution/hint stay coherent with body
        sp["solution_text"] = substitute_variables(sp["solution_text"], values)
        sp["hint_text"] = substitute_variables(sp["hint_text"], values)
    return data
```

Test (`apps/api/tests/test_subpart_substitution.py`): hitting
`/api/v1/assignments/<id>/` as a student returns substituted solution +

### What could go wrong

- A handful of legacy expressions reference custom helpers we don't yet
  allowlist. The `_repl` exception fallback prevents a crash; we surface
  unknown identifiers in a `--report-unknowns` flag on the importer to
  triage them.
- Image filenames may not match `<subpart_id>.png` exactly for every
  chapter. Pattern-match `<subpart_id>.*` and log misses.
- Re-import replaces subparts wholesale (existing P8 behaviour). Any
  ad-hoc edits made via Django admin after import will be lost — but
  none exist yet, so safe.
